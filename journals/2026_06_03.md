- 系统：CentOS 7.5.1804，Linux内核版本3.10.0-862.el7.x86_64
- 作业调度系统：Torque 4.2.10
- 提交任务脚本如下：适用于点节点，跨节点没有测试，脚本与模型放置在同一目录，结果自动保存在与模型同名的文件夹中
  
  ``` bash
  #!/bin/sh
  
  #PBS -N PY
  #PBS -l walltime=4:00:00
  #PBS -q batch
  #PBS -e ${PBS_JOBNAME}.err
  #PBS -o ${PBS_JOBNAME}.out
  #PBS -j n
  #PBS -l nodes=1:ppn=48 
  
  cd $PBS_O_WORKDIR
  
  np=48
  inputfile="water_tank"
  mkdir $inputfile
  cd $inputfile
  
  export LD_LIBRARY_PATH=/apps/gsl/lib:$LD_LIBRARY_PATH
  
  CONTAINER="$HOME/Containers/comsol64.sif"
  comsol="/usr/local/comsol64/multiphysics/bin/comsol"
  singularity run $CONTAINER \
    $comsol batch -np $np \
    -inputfile ../${inputfile}.mph \
    -outputfile ${inputfile}_result.mph \
    -batchlog batch_COMSOL__$$.log
  ```
